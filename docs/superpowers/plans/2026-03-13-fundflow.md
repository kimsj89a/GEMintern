# FundFlow Implementation Plan

**Goal:** Build a standalone LP funding tracker web app with team collaboration, fund/LP pipeline management, meeting logs, action items, and activity feed.

**Architecture:** FastAPI + SQLite backend serves React SPA. Team-scoped multi-tenancy via JWT auth with invite codes. All data scoped by team_id. React Router for URL-based navigation.

Follow GEMintern's database.py pattern: thread-local connections, WAL mode, PRAGMA foreign_keys=ON, context manager `get_db()`, PBKDF2-SHA256 password hashing. Schema includes all 9 tables from the spec.

```python
"""SQLite database for FundFlow — LP funding tracker."""
import os
import sqlite3
import hashlib
import secrets
import threading
from contextlib import contextmanager

DB_PATH = os.environ.get("DB_PATH", os.path.join(
    os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "data", "fundflow.db"
))

_local = threading.local()


def _get_conn() -> sqlite3.Connection:
    if not hasattr(_local, "conn") or _local.conn is None:
        os.makedirs(os.path.dirname(DB_PATH), exist_ok=True)
        _local.conn = sqlite3.connect(DB_PATH)
        _local.conn.row_factory = sqlite3.Row
        _local.conn.execute("PRAGMA journal_mode=WAL")
        _local.conn.execute("PRAGMA foreign_keys=ON")
    return _local.conn


@contextmanager
def get_db():
    conn = _get_conn()
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise


def hash_password(password: str, salt: bytes | None = None) -> tuple[str, str]:
    if salt is None:
        salt = secrets.token_bytes(16)
    dk = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 100_000)
    return dk.hex(), salt.hex()


def verify_password(password: str, stored_hash: str, salt_hex: str) -> bool:
    salt = bytes.fromhex(salt_hex)
    dk = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 100_000)
    return dk.hex() == stored_hash


def init_db():
    """Create all tables. Safe to call multiple times (IF NOT EXISTS)."""
    with get_db() as conn:
        conn.executescript("""
            CREATE TABLE IF NOT EXISTS teams (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            );

            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE NOT NULL,
                password_hash TEXT NOT NULL,
                password_salt TEXT NOT NULL,
                team_id INTEGER NOT NULL,
                role TEXT NOT NULL DEFAULT 'member',
                display_name TEXT NOT NULL DEFAULT '',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (team_id) REFERENCES teams(id)
            );

            CREATE TABLE IF NOT EXISTS invite_codes (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                code TEXT UNIQUE NOT NULL,
                team_id INTEGER NOT NULL,
                created_by INTEGER NOT NULL,
                used_by INTEGER NULL,
                used_at TIMESTAMP NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (team_id) REFERENCES teams(id),
                FOREIGN KEY (created_by) REFERENCES users(id),
                FOREIGN KEY (used_by) REFERENCES users(id)
            );

            CREATE TABLE IF NOT EXISTS funds (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                team_id INTEGER NOT NULL,
                name TEXT NOT NULL,
                target_amount INTEGER NOT NULL DEFAULT 0,
                status TEXT NOT NULL DEFAULT 'active',
                description TEXT DEFAULT '',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (team_id) REFERENCES teams(id)
            );

            CREATE TABLE IF NOT EXISTS lps (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                team_id INTEGER NOT NULL,
                name TEXT NOT NULL,
                type TEXT NOT NULL DEFAULT 'other',
                contact_name TEXT DEFAULT '',
                contact_info TEXT DEFAULT '',
                notes TEXT DEFAULT '',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (team_id) REFERENCES teams(id),
                UNIQUE(team_id, name)
            );

            CREATE TABLE IF NOT EXISTS fund_lps (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                fund_id INTEGER NOT NULL,
                lp_id INTEGER NOT NULL,
                status TEXT NOT NULL DEFAULT 'contacted',
                commit_amount INTEGER NOT NULL DEFAULT 0,
                paid_amount INTEGER NOT NULL DEFAULT 0,
                assigned_to INTEGER NULL,
                notes TEXT DEFAULT '',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (fund_id) REFERENCES funds(id) ON DELETE CASCADE,
                FOREIGN KEY (lp_id) REFERENCES lps(id),
                FOREIGN KEY (assigned_to) REFERENCES users(id),
                UNIQUE(fund_id, lp_id)
            );

            CREATE TABLE IF NOT EXISTS activities (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                team_id INTEGER NOT NULL,
                user_id INTEGER NOT NULL,
                fund_lp_id INTEGER NULL,
                fund_id INTEGER NULL,
                action TEXT NOT NULL,
                detail TEXT DEFAULT '{}',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (team_id) REFERENCES teams(id),
                FOREIGN KEY (user_id) REFERENCES users(id),
                FOREIGN KEY (fund_lp_id) REFERENCES fund_lps(id) ON DELETE SET NULL,
                FOREIGN KEY (fund_id) REFERENCES funds(id) ON DELETE CASCADE
            );

            CREATE TABLE IF NOT EXISTS meetings (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                fund_lp_id INTEGER NOT NULL,
                date DATE NOT NULL,
                summary TEXT DEFAULT '',
                next_action TEXT DEFAULT '',
                created_by INTEGER NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (fund_lp_id) REFERENCES fund_lps(id) ON DELETE CASCADE,
                FOREIGN KEY (created_by) REFERENCES users(id)
            );

            CREATE TABLE IF NOT EXISTS action_items (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                team_id INTEGER NOT NULL,
                fund_lp_id INTEGER NULL,
                title TEXT NOT NULL,
                due_date DATE NULL,
                assigned_to INTEGER NULL,
                completed BOOLEAN NOT NULL DEFAULT 0,
                completed_at TIMESTAMP NULL,
                created_by INTEGER NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (team_id) REFERENCES teams(id),
                FOREIGN KEY (fund_lp_id) REFERENCES fund_lps(id) ON DELETE CASCADE,
                FOREIGN KEY (assigned_to) REFERENCES users(id),
                FOREIGN KEY (created_by) REFERENCES users(id)
            );
        """)
```

JWT helpers: `create_token`, `decode_token`, `get_current_user` (FastAPI Depends), `require_admin` (FastAPI Depends). Same pattern as GEMintern but with team_id and role.

```python
"""JWT authentication utilities for FundFlow."""
import os
from datetime import datetime, timedelta, timezone

import jwt
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials

from backend.database import get_db

JWT_SECRET = os.environ.get("JWT_SECRET", "dev-secret-change-in-production")
JWT_ALGORITHM = "HS256"
JWT_EXPIRY_HOURS = int(os.environ.get("JWT_EXPIRY_HOURS", "24"))

_bearer_scheme = HTTPBearer()


def create_token(user_id: int, username: str, team_id: int, role: str) -> str:
    payload = {
        "sub": str(user_id),
        "username": username,
        "team_id": team_id,
        "role": role,
        "exp": datetime.now(timezone.utc) + timedelta(hours=JWT_EXPIRY_HOURS),
    }
    return jwt.encode(payload, JWT_SECRET, algorithm=JWT_ALGORITHM)


def decode_token(token: str) -> dict:
    try:
        return jwt.decode(token, JWT_SECRET, algorithms=[JWT_ALGORITHM])
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Token expired")
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid token")


async def get_current_user(
    creds: HTTPAuthorizationCredentials = Depends(_bearer_scheme),
) -> dict:
    """Decode JWT and return full user dict from DB."""
    payload = decode_token(creds.credentials)
    with get_db() as conn:
        user = conn.execute(
            "SELECT id, username, display_name, team_id, role FROM users WHERE id = ?",
            (int(payload["sub"]),),
        ).fetchone()
    if not user:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="User not found")
    return {
        "id": user["id"],
        "username": user["username"],
        "display_name": user["display_name"],
        "team_id": user["team_id"],
        "role": user["role"],
    }


async def require_admin(user: dict = Depends(get_current_user)) -> dict:
    """Dependency that ensures the current user has admin role."""
    if user["role"] != "admin":
        raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="Admin required")
    return user
```

Auth API router with register (two modes), login, me, update profile.

```python
"""Authentication API routes for FundFlow."""
import secrets

from fastapi import APIRouter, Depends, HTTPException, status

from backend.database import get_db, hash_password, verify_password
from backend.auth import create_token, get_current_user, require_admin
from backend.api_models import RegisterRequest, LoginRequest, UpdateProfileRequest

router = APIRouter(prefix="/auth", tags=["auth"])


@router.post("/register")
async def register(req: RegisterRequest):
    """Register a new user.

    Two modes:
    - team_name provided: creates a new team + admin user
    - invite_code provided: joins existing team as member
    Exactly one of team_name or invite_code must be provided.
    """
    if req.team_name and req.invite_code:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Provide either team_name or invite_code, not both",
        )
    if not req.team_name and not req.invite_code:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Provide either team_name (create team) or invite_code (join team)",
        )

    pw_hash, salt = hash_password(req.password)

    if req.team_name:
        # --- Team creation mode ---
        with get_db() as conn:
            existing = conn.execute(
                "SELECT id FROM users WHERE username = ?", (req.username,)
            ).fetchone()
            if existing:
                raise HTTPException(
                    status_code=status.HTTP_409_CONFLICT, detail="Username already taken"
                )
            cur = conn.execute(
                "INSERT INTO teams (name) VALUES (?)", (req.team_name,)
            )
            team_id = cur.lastrowid
            cur = conn.execute(
                "INSERT INTO users (username, password_hash, password_salt, team_id, role, display_name) "
                "VALUES (?, ?, ?, ?, 'admin', ?)",
                (req.username, pw_hash, salt, team_id, req.display_name),
            )
            user_id = cur.lastrowid

        token = create_token(user_id, req.username, team_id, "admin")
        return {
            "token": token,
            "user": {
                "id": user_id,
                "username": req.username,
                "display_name": req.display_name,
                "team_id": team_id,
                "role": "admin",
            },
        }

    else:
        # --- Team join mode (invite_code) ---
        with get_db() as conn:
            code_row = conn.execute(
                "SELECT id, team_id FROM invite_codes WHERE code = ? AND used_by IS NULL",
                (req.invite_code,),
            ).fetchone()
            if not code_row:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="Invalid or already used invite code",
                )

            existing = conn.execute(
                "SELECT id FROM users WHERE username = ?", (req.username,)
            ).fetchone()
            if existing:
                raise HTTPException(
                    status_code=status.HTTP_409_CONFLICT, detail="Username already taken"
                )

            team_id = code_row["team_id"]
            cur = conn.execute(
                "INSERT INTO users (username, password_hash, password_salt, team_id, role, display_name) "
                "VALUES (?, ?, ?, ?, 'member', ?)",
                (req.username, pw_hash, salt, team_id, req.display_name),
            )
            user_id = cur.lastrowid
            conn.execute(
                "UPDATE invite_codes SET used_by = ?, used_at = CURRENT_TIMESTAMP WHERE id = ?",
                (user_id, code_row["id"]),
            )

        token = create_token(user_id, req.username, team_id, "member")
        return {
            "token": token,
            "user": {
                "id": user_id,
                "username": req.username,
                "display_name": req.display_name,
                "team_id": team_id,
                "role": "member",
            },
        }


@router.post("/login")
async def login(req: LoginRequest):
    """Authenticate user and return JWT."""
    with get_db() as conn:
        user = conn.execute(
            "SELECT id, username, password_hash, password_salt, team_id, role, display_name "
            "FROM users WHERE username = ?",
            (req.username,),
        ).fetchone()
    if not user or not verify_password(req.password, user["password_hash"], user["password_salt"]):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid credentials"
        )
    token = create_token(user["id"], user["username"], user["team_id"], user["role"])
    return {
        "token": token,
        "user": {
            "id": user["id"],
            "username": user["username"],
            "display_name": user["display_name"],
            "team_id": user["team_id"],
            "role": user["role"],
        },
    }


@router.get("/me")
async def me(user: dict = Depends(get_current_user)):
    """Return current user info."""
    return user


@router.patch("/me")
async def update_profile(req: UpdateProfileRequest, user: dict = Depends(get_current_user)):
    """Update current user's display_name and/or password."""
    if req.display_name is None and req.password is None:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Nothing to update",
        )

    with get_db() as conn:
        if req.display_name is not None:
            conn.execute(
                "UPDATE users SET display_name = ? WHERE id = ?",
                (req.display_name, user["id"]),
            )
        if req.password is not None:
            pw_hash, salt = hash_password(req.password)
            conn.execute(
                "UPDATE users SET password_hash = ?, password_salt = ? WHERE id = ?",
                (pw_hash, salt, user["id"]),
            )

    # Re-fetch updated user
    with get_db() as conn:
        updated = conn.execute(
            "SELECT id, username, display_name, team_id, role FROM users WHERE id = ?",
            (user["id"],),
        ).fetchone()
    return dict(updated)
```

```python
import secrets
import string
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from typing import Optional

from backend.auth import get_current_user
from backend.database import get_db

router = APIRouter(tags=["team"])


# ── Pydantic models ──────────────────────────────────────────────

class TeamUpdate(BaseModel):
    name: str

class MemberOut(BaseModel):
    id: int
    username: str
    display_name: str
    role: str

class RoleUpdate(BaseModel):
    role: str  # 'admin' or 'member'

class InviteCodeOut(BaseModel):
    id: int
    code: str
    created_by: int
    used_by: Optional[int] = None
    used_by_username: Optional[str] = None
    used_at: Optional[str] = None
    created_at: str


# ── Helpers ───────────────────────────────────────────────────────

def _require_admin(user: dict):
    if user["role"] != "admin":
        raise HTTPException(status_code=403, detail="Admin access required")


# ── Endpoints ─────────────────────────────────────────────────────

@router.patch("")
async def update_team(body: TeamUpdate, user: dict = Depends(get_current_user)):
    """Update team name (admin only)."""
    _require_admin(user)
    with get_db() as conn:
        conn.execute("UPDATE teams SET name = ? WHERE id = ?", (body.name, user["team_id"]))
    return {"ok": True, "name": body.name}


@router.get("/members")
async def list_members(user: dict = Depends(get_current_user)):
    """List team members. Also returns team_name for convenience."""
    with get_db() as conn:
        team = conn.execute("SELECT name FROM teams WHERE id = ?", (user["team_id"],)).fetchone()
        team_name = team["name"] if team else ""
        rows = conn.execute(
            "SELECT id, username, display_name, role FROM users WHERE team_id = ? ORDER BY id",
            (user["team_id"],),
        ).fetchall()
    return {
        "team_name": team_name,
        "members": [
            MemberOut(id=r["id"], username=r["username"], display_name=r["display_name"], role=r["role"])
            for r in rows
        ],
    }


@router.patch("/members/{member_id}")
async def update_member_role(member_id: int, body: RoleUpdate, user: dict = Depends(get_current_user)):
    """Change a member's role (admin only)."""
    _require_admin(user)
    if body.role not in ("admin", "member"):
        raise HTTPException(status_code=400, detail="Role must be 'admin' or 'member'")

    with get_db() as conn:
        target = conn.execute(
            "SELECT id, team_id FROM users WHERE id = ?", (member_id,)
        ).fetchone()
        if not target or target["team_id"] != user["team_id"]:
            raise HTTPException(status_code=404, detail="Member not found")

        conn.execute("UPDATE users SET role = ? WHERE id = ?", (body.role, member_id))
    return {"ok": True}


@router.post("/invite-codes")
async def create_invite_code(user: dict = Depends(get_current_user)):
    """Generate a random 8-char alphanumeric invite code (admin only)."""
    _require_admin(user)
    alphabet = string.ascii_uppercase + string.digits
    code = "".join(secrets.choice(alphabet) for _ in range(8))

    with get_db() as conn:
        conn.execute(
            "INSERT INTO invite_codes (code, team_id, created_by) VALUES (?, ?, ?)",
            (code, user["team_id"], user["id"]),
        )
    return {"code": code}


@router.get("/invite-codes")
async def list_invite_codes(user: dict = Depends(get_current_user)):
    """List invite codes with used_by info (admin only)."""
    _require_admin(user)
    with get_db() as conn:
        rows = conn.execute(
            """
            SELECT ic.id, ic.code, ic.created_by, ic.used_by, ic.used_at, ic.created_at,
                   u.username AS used_by_username
            FROM invite_codes ic
            LEFT JOIN users u ON ic.used_by = u.id
            WHERE ic.team_id = ?
            ORDER BY ic.created_at DESC
            """,
            (user["team_id"],),
        ).fetchall()
    return [
        InviteCodeOut(
            id=r["id"],
            code=r["code"],
            created_by=r["created_by"],
            used_by=r["used_by"],
            used_by_username=r["used_by_username"],
            used_at=r["used_at"],
            created_at=r["created_at"],
        )
        for r in rows
    ]
```

```python
@router.patch("/action-items/{item_id}")
async def update_action_item(item_id: int, body: ActionItemUpdate, user: dict = Depends(get_current_user)):
    """Update action item. If completing, creates 'action_item_completed' activity."""
    with get_db() as conn:
        old = conn.execute(
            "SELECT * FROM action_items WHERE id = ? AND team_id = ?",
            (item_id, user["team_id"]),
        ).fetchone()
        if not old:
            raise HTTPException(status_code=404, detail="Action item not found")

        updates = {}
        if body.title is not None:
            updates["title"] = body.title
        if body.due_date is not None:
            updates["due_date"] = body.due_date
        if body.assigned_to is not None:
            if body.assigned_to == 0:
                updates["assigned_to"] = None  # 0 means "unassign"
            else:
                updates["assigned_to"] = body.assigned_to
        if body.completed is not None:
            updates["completed"] = 1 if body.completed else 0
            if body.completed:
                updates["completed_at"] = datetime.now(timezone.utc).isoformat()
            else:
                updates["completed_at"] = None

        if not updates:
            raise HTTPException(status_code=400, detail="No fields to update")

        set_clause = ", ".join(f"{k} = ?" for k in updates)
        values = list(updates.values()) + [item_id]
        conn.execute(f"UPDATE action_items SET {set_clause} WHERE id = ?", values)

        # Log completion activity
        if body.completed and not old["completed"]:
            fund_id = None
            if old["fund_lp_id"]:
                fl = conn.execute(
                    "SELECT fund_id FROM fund_lps WHERE id = ?", (old["fund_lp_id"],)
                ).fetchone()
                fund_id = fl["fund_id"] if fl else None

            _log_activity(conn, user["team_id"], user["id"], old["fund_lp_id"], fund_id,
                          "action_item_completed", {"title": body.title or old["title"]})

        row = conn.execute(
            """SELECT ai.id, ai.team_id, ai.fund_lp_id, ai.title, ai.due_date,
                      ai.assigned_to, ai.completed, ai.completed_at, ai.created_by, ai.created_at,
                      u.display_name AS assigned_to_name,
                      f.name AS fund_name,
                      l.name AS lp_name
               FROM action_items ai
               LEFT JOIN users u ON ai.assigned_to = u.id
               LEFT JOIN fund_lps fl ON ai.fund_lp_id = fl.id
               LEFT JOIN funds f ON fl.fund_id = f.id
               LEFT JOIN lps l ON fl.lp_id = l.id
               WHERE ai.id = ?""",
            (item_id,),
        ).fetchone()
    return ActionItemOut(**dict(row))
```

## Key design decisions

1. **Context manager pattern**: All route handlers use `with get_db() as conn:` (matching Chunk 1's `auth_routes.py`), NOT FastAPI `Depends(get_db)`. This ensures proper transaction commit/rollback semantics.

2. **`_log_activity()` helper**: Centralized in `api_routes.py`, called inline after every mutation that should be tracked. Accepts `detail_dict` which gets `json.dumps()`'d into the `detail` TEXT column.

3. **Team ownership checks**: Every endpoint uses `_get_fund_or_404()`, `_get_lp_or_404()`, or `_get_fund_lp_or_404()` to verify the requesting user's `team_id` matches the resource.

4. **Cascade delete strategy**: `DELETE /funds/:id` manually cascades through fund_lps, meetings, action_items, activities. `DELETE /fund-lps/:id` cascades meetings and action_items but preserves activities. `DELETE /lps/:id` returns 400 if linked.

5. **Activity auto-logging**: `PATCH /fund-lps/:id` compares old vs new values and creates separate activity records for `status_change`, `amount_changed` (commit and paid tracked independently), and `note_added`. Same-value updates create no activity.

6. **Dashboard aggregation**: Single endpoint performs three queries — funds with `SUM()` aggregates, incomplete action items with JOINed names, and last 20 activities with JOINed names.

7. **Test pattern**: All tests use the `client` fixture from `conftest.py` (Chunk 1), with `resp.json()["token"]` to extract the auth token.

```ts
import { create } from 'zustand';

export interface AuthUser {
  id: number;
  username: string;
  display_name: string;
  team_id: number;
  role: 'admin' | 'member';
}

interface AuthState {
  token: string | null;
  user: AuthUser | null;
  setAuth: (token: string, user: AuthUser) => void;
  logout: () => void;
}

const STORAGE_KEY = 'fundflow_auth';

function loadFromStorage(): { token: string | null; user: AuthUser | null } {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (raw) {
      const { token, user } = JSON.parse(raw);
      return { token, user };
    }
  } catch { /* ignore */ }
  return { token: null, user: null };
}

const initial = loadFromStorage();

export const useAuthStore = create<AuthState>((set) => ({
  token: initial.token,
  user: initial.user,
  setAuth: (token, user) => {
    localStorage.setItem(STORAGE_KEY, JSON.stringify({ token, user }));
    set({ token, user });
  },
  logout: () => {
    localStorage.removeItem(STORAGE_KEY);
    set({ token: null, user: null });
  },
}));
```

```ts
import { useAuthStore, AuthUser } from '../stores/authStore';

const BASE = '/api';

function getAuthHeaders(): Record<string, string> {
  const token = useAuthStore.getState().token;
  if (token) return { Authorization: `Bearer ${token}` };
  return {};
}

async function request<T>(path: string, options?: RequestInit): Promise<T> {
  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    ...getAuthHeaders(),
  };
  const res = await fetch(`${BASE}${path}`, {
    ...options,
    headers: { ...headers, ...(options?.headers as Record<string, string>) },
  });
  if (res.status === 401) {
    useAuthStore.getState().logout();
    throw new Error('Unauthorized');
  }
  if (!res.ok) {
    const body = await res.json().catch(() => ({}));
    throw new Error(body.detail || `API error: ${res.status}`);
  }
  if (res.status === 204) return undefined as T;
  return res.json();
}
```

```tsx
function getDday(dueDate: string | null): { label: string; urgency: 'red' | 'yellow' | 'normal' } {
  if (!dueDate) return { label: '', urgency: 'normal' };
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const due = new Date(dueDate);
  due.setHours(0, 0, 0, 0);
  const diff = Math.ceil((due.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
  if (diff < 0) return { label: `D+${Math.abs(diff)}`, urgency: 'red' };
  if (diff === 0) return { label: 'D-Day', urgency: 'red' };
  if (diff <= 2) return { label: `D-${diff}`, urgency: 'red' };
  if (diff <= 7) return { label: `D-${diff}`, urgency: 'yellow' };
  return { label: `D-${diff}`, urgency: 'normal' };
}

export default function ActionItemList({ items, onUpdate }: ActionItemListProps) {
  const [togglingId, setTogglingId] = useState<number | null>(null);
  const [showAddForm, setShowAddForm] = useState(false);
  const [newTitle, setNewTitle] = useState('');
  const [newDueDate, setNewDueDate] = useState('');
  const [adding, setAdding] = useState(false);

  const handleAdd = async () => {
    if (!newTitle.trim()) return;
    setAdding(true);
    try {
      await api.createActionItem({
        title: newTitle.trim(),
        due_date: newDueDate || undefined,
      });
      setNewTitle('');
      setNewDueDate('');
      setShowAddForm(false);
      onUpdate();
    } catch (err) {
      console.error('Failed to create action item:', err);
    } finally {
      setAdding(false);
    }
  };

  const handleDelete = async (id: number) => {
    try {
      await api.deleteActionItem(id);
      onUpdate();
    } catch (err) {
      console.error('Failed to delete action item:', err);
    }
  };

  const sorted = [...items].sort((a, b) => {
    // Completed items go to the bottom
    if (a.completed !== b.completed) return a.completed ? 1 : -1;
    // Then sort by due_date (nulls last)
    if (!a.due_date && !b.due_date) return 0;
    if (!a.due_date) return 1;
    if (!b.due_date) return -1;
    return new Date(a.due_date).getTime() - new Date(b.due_date).getTime();
  });

  const handleToggle = async (item: ActionItem) => {
    setTogglingId(item.id);
    try {
      await api.updateActionItem(item.id, { completed: !item.completed });
      onUpdate();
    } catch (err) {
      console.error('Failed to toggle action item:', err);
    } finally {
      setTogglingId(null);
    }
  };

  if (items.length === 0) {
    return (

            value={newTitle}
            onChange={(e) => setNewTitle(e.target.value)}
            placeholder="할일 제목"
            className="flex-1 px-2 py-1 text-sm bg-[var(--color-bg)] input-ring"
            autoFocus
            onKeyDown={(e) => { if (e.key === 'Enter') handleAdd(); if (e.key === 'Escape') setShowAddForm(false); }}

            className="px-2 py-1 text-sm bg-[var(--color-bg)] input-ring"

          </button>
        </div>
      ) : (
```

```tsx
function formatTime(dateStr: string): string {
  const date = new Date(dateStr);
  const now = new Date();
  const diffMs = now.getTime() - date.getTime();
  const diffMin = Math.floor(diffMs / 60000);
  if (diffMin < 1) return '방금 전';
  if (diffMin < 60) return `${diffMin}분 전`;
  const diffHour = Math.floor(diffMin / 60);
  if (diffHour < 24) return `${diffHour}시간 전`;
  const diffDay = Math.floor(diffHour / 24);
  if (diffDay < 7) return `${diffDay}일 전`;
  return date.toLocaleDateString('ko-KR', { month: 'short', day: 'numeric' });
}

}

const AVATAR_COLORS = [
```

```tsx
import { useState, useMemo } from 'react';
import type { FundLP, FundLpStatus } from '../api/client';

interface LpTableProps {
  fundLps: FundLP[];
  onClickLp: (fundLp: FundLP) => void;
}

const STATUS_LABELS: Record<FundLpStatus, string> = {
  contacted: '접촉',
  interested: '관심표명',
  negotiating: '조건협의',
  committed: '확약',
  paid: '납입',
  dropped: 'Drop',
};

const LP_TYPE_LABELS: Record<string, string> = {
  insurance: '보험',
  securities: '증권사',
  bank: '은행',
  pension: '연기금',
  mutual_aid: '공제회',
  other: '기타',
};

type SortKey = 'lp_name' | 'lp_type' | 'status' | 'commit_amount' | 'paid_amount' | 'assigned_to_name';
type SortDir = 'asc' | 'desc';

const STATUS_ORDER: Record<string, number> = {
  contacted: 0,
  interested: 1,
  negotiating: 2,
  committed: 3,
  paid: 4,
  dropped: 5,
};

export default function LpTable({ fundLps, onClickLp }: LpTableProps) {
  const [sortKey, setSortKey] = useState<SortKey>('lp_name');
  const [sortDir, setSortDir] = useState<SortDir>('asc');
  const [statusFilter, setStatusFilter] = useState<FundLpStatus | ''>('');

  const handleSort = (key: SortKey) => {
    if (sortKey === key) {
      setSortDir(sortDir === 'asc' ? 'desc' : 'asc');
    } else {
      setSortKey(key);
      setSortDir('asc');
    }
  };

  const filtered = useMemo(() => {
    let result = [...fundLps];
    if (statusFilter) {
      result = result.filter((lp) => lp.status === statusFilter);
    }
    return result;
  }, [fundLps, statusFilter]);

  const sorted = useMemo(() => {
    return [...filtered].sort((a, b) => {
      let cmp = 0;
      switch (sortKey) {
        case 'lp_name':
          cmp = (a.lp_name || '').localeCompare(b.lp_name || '');
          break;
        case 'lp_type':
          cmp = (a.lp_type || '').localeCompare(b.lp_type || '');
          break;
        case 'status':
          cmp = (STATUS_ORDER[a.status] ?? 99) - (STATUS_ORDER[b.status] ?? 99);
          break;
        case 'commit_amount':
          cmp = a.commit_amount - b.commit_amount;
          break;
        case 'paid_amount':
          cmp = a.paid_amount - b.paid_amount;
          break;
        case 'assigned_to_name':
          cmp = (a.assigned_to_name || '').localeCompare(b.assigned_to_name || '');
          break;
      }
      return sortDir === 'asc' ? cmp : -cmp;
    });
  }, [filtered, sortKey, sortDir]);

  const SortHeader = ({ label, field }: { label: string; field: SortKey }) => (
    <th
      className="px-4 py-3 text-left text-[11px] font-semibold text-[var(--color-text-tertiary)] uppercase tracking-wider cursor-pointer hover:text-[var(--color-text-secondary)] select-none"
      onClick={() => handleSort(field)}
    >
      <span className="inline-flex items-center gap-1">
        {label}
        {sortKey === field && (
          <svg className={`w-3 h-3 transition-transform ${sortDir === 'desc' ? 'rotate-180' : ''}`} fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
            <path strokeLinecap="round" strokeLinejoin="round" d="M5 15l7-7 7 7" />
          </svg>
        )}
      </span>
    </th>
  );

  return (
    <div>
      {/* Filter */}
      <div className="px-1 pb-3">
        <select
          value={statusFilter}
          onChange={(e) => setStatusFilter(e.target.value as FundLpStatus | '')}
          className="text-sm px-3 py-1.5 bg-[var(--color-bg)] input-ring cursor-pointer"
        >
          <option value="">전체 상태</option>
          {(Object.entries(STATUS_LABELS) as [FundLpStatus, string][]).map(([value, label]) => (
            <option key={value} value={value}>{label}</option>
          ))}
        </select>
      </div>

      {/* Table */}
      <div className="overflow-x-auto">
        <table className="w-full">
          <thead>
            <tr className="border-b border-[var(--color-border)]">
              <SortHeader label="LP명" field="lp_name" />
              <SortHeader label="유형" field="lp_type" />
              <SortHeader label="상태" field="status" />
              <SortHeader label="커밋금액" field="commit_amount" />
              <SortHeader label="납입금액" field="paid_amount" />
              <SortHeader label="담당자" field="assigned_to_name" />
              <th className="px-4 py-3 text-left text-[11px] font-semibold text-[var(--color-text-tertiary)] uppercase tracking-wider">
                최근활동
              </th>
            </tr>
          </thead>
          <tbody>
            {sorted.length === 0 ? (
              <tr>
                <td colSpan={7} className="px-4 py-12 text-center text-sm text-[var(--color-text-tertiary)]">
                  {statusFilter ? '해당 상태의 LP가 없습니다' : 'LP가 없습니다. LP를 추가해주세요.'}
                </td>
              </tr>
            ) : (
              sorted.map((fundLp) => (
                <tr
                  key={fundLp.id}
                  onClick={() => onClickLp(fundLp)}
                  className="border-b border-[var(--color-border-subtle)] hover:bg-[var(--color-bg-subtle)] transition-colors cursor-pointer"
                >
                  <td className="px-4 py-3 text-sm font-medium text-[var(--color-text)]">
                    {fundLp.lp_name || '-'}
                  </td>
                  <td className="px-4 py-3 text-sm text-[var(--color-text-secondary)]">
                    {LP_TYPE_LABELS[fundLp.lp_type || ''] || fundLp.lp_type || '-'}
                  </td>
                  <td className="px-4 py-3">
                    <span className={`inline-block px-2.5 py-1 text-[11px] font-semibold rounded-full badge-${fundLp.status}`}>
                      {STATUS_LABELS[fundLp.status]}
                    </span>
                  </td>
                  <td className="px-4 py-3 text-sm text-[var(--color-text-secondary)] tabular-nums">
                    {fundLp.commit_amount > 0 ? `${fundLp.commit_amount}억` : '-'}
                  </td>
                  <td className="px-4 py-3 text-sm text-[var(--color-text-secondary)] tabular-nums">
                    {fundLp.paid_amount > 0 ? `${fundLp.paid_amount}억` : '-'}
                  </td>
                  <td className="px-4 py-3 text-sm text-[var(--color-text-secondary)]">
                    {fundLp.assigned_to_name || '-'}
                  </td>
                  <td className="px-4 py-3 text-[12px] text-[var(--color-text-tertiary)]">
                    {fundLp.updated_at
                      ? new Date(fundLp.updated_at).toLocaleDateString('ko-KR', { month: 'short', day: 'numeric' })
                      : '-'}
                  </td>
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>
    </div>
  );
}
```

```tsx
function DraggableCard({

  onClick,

      }}

    <div className="glass-card-elevated p-3 w-[184px] shadow-lg rotate-2">
      <p className="text-sm font-semibold text-[var(--color-text)] truncate">{fundLp.lp_name || '-'}</p>
      {fundLp.commit_amount > 0 && (
        <p className="text-[12px] text-[var(--color-text-secondary)] mt-1 tabular-nums">
          {fundLp.commit_amount}억
        </p>
      )}
      {fundLp.assigned_to_name && (
        <p className="text-[11px] text-[var(--color-text-tertiary)] mt-1 truncate">

  );

export default function KanbanBoard({ fundLps, onStatusChange, onClickLp }: KanbanBoardProps) {
  const [activeItem, setActiveItem] = useState<FundLP | null>(null);

  const sensors = useSensors(
    useSensor(PointerSensor, {
      activationConstraint: { distance: 5 },
    })
  );

  const handleDragStart = (event: DragStartEvent) => {
    const item = fundLps.find((lp) => lp.id === event.active.id);
    if (item) setActiveItem(item);
  };

  const handleDragEnd = (event: DragEndEvent) => {
    setActiveItem(null);
    const { active, over } = event;
    if (!over) return;

    const fundLpId = active.id as number;
    const newStatus = over.id as string;
    const current = fundLps.find((lp) => lp.id === fundLpId);
    if (!current || current.status === newStatus) return;

    onStatusChange(fundLpId, newStatus);
  };

  // Group by status
  const grouped: Record<FundLpStatus, FundLP[]> = {
    contacted: [],
    interested: [],
    negotiating: [],
    committed: [],
    paid: [],
    dropped: [],
  };
  for (const lp of fundLps) {
    if (grouped[lp.status]) {
      grouped[lp.status].push(lp);
    }
  }

  return (
    <DndContext
      sensors={sensors}
      onDragStart={handleDragStart}
      onDragEnd={handleDragEnd}
    >
      <div className="flex gap-3 h-full overflow-x-auto pb-4">
        {COLUMNS.map((col) => (
          <DroppableColumn
            key={col.status}
            status={col.status}
            label={col.label}
            items={grouped[col.status]}
            onClickLp={onClickLp}
          />
        ))}
      </div>

      <DragOverlay dropAnimation={null}>
        {activeItem && <OverlayCard fundLp={activeItem} />}
      </DragOverlay>
    </DndContext>
  );
}
```

```tsx
                    <p className="text-[11px] text-[var(--color-text-tertiary)]">@{member.username}</p>
                  </div>
                </div>
                <div className="flex items-center gap-2">
                  {member.id === user?.id ? (
                    <span className="text-[11px] px-2.5 py-1 rounded-full bg-[var(--color-primary)]/10 text-[var(--color-primary)] font-semibold">
                      {member.role === 'admin' ? '관리자' : '멤버'} (나)
                    </span>
                  ) : (
                    <select
                      value={member.role}
                      onChange={(e) => handleRoleChange(member.id, e.target.value as 'admin' | 'member')}
                      className="text-[12px] px-2 py-1 bg-[var(--color-bg)] input-ring cursor-pointer"
                    >
                      <option value="admin">관리자</option>
                      <option value="member">멤버</option>
                    </select>
                  )}
                </div>
              </div>
            ))}
          </div>

          {/* Invite codes */}
          <div className="border-t border-[var(--color-border)] pt-4">
            <div className="flex items-center justify-between mb-3">
              <h3 className="text-sm font-semibold text-[var(--color-text)]">초대 코드</h3>
              <button
                onClick={handleGenerateCode}
                disabled={generatingCode}
                className="px-3 py-1.5 text-[12px] btn-primary cursor-pointer"
              >
                {generatingCode ? '생성 중...' : '새 코드 생성'}
              </button>
            </div>

            {inviteCodes.length === 0 ? (
              <p className="text-sm text-[var(--color-text-tertiary)]">생성된 초대 코드가 없습니다</p>
            ) : (
              <div className="space-y-2">
                {inviteCodes.map((ic) => (
                  <div
                    key={ic.id}
                    className="flex items-center justify-between px-3 py-2 rounded-lg bg-[var(--color-bg-subtle)]"
                  >
                    <div className="flex items-center gap-3">
                      <code className="text-sm font-mono font-semibold text-[var(--color-text)]">{ic.code}</code>
                      {ic.used_by ? (
                        <span className="text-[11px] text-[var(--color-text-tertiary)]">사용됨</span>
                      ) : (
                        <span className="text-[11px] text-[var(--color-accent)] font-medium">미사용</span>
                      )}
                    </div>
                    {!ic.used_by && (
                      <button
                        onClick={() => handleCopyCode(ic.code)}
                        className="text-[12px] text-[var(--color-primary)] hover:underline cursor-pointer"
                      >
                        {copiedCode === ic.code ? '복사됨!' : '복사'}
                      </button>
                    )}
                  </div>
                ))}
              </div>
            )}
          </div>
        </section>
      )}

      {/* ──── 내 정보 ──── */}
      <section className="glass-card-elevated p-6">
        <h2 className="text-base font-bold text-[var(--color-text)] mb-4">내 정보</h2>
        <form onSubmit={handleSaveProfile} className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-[var(--color-text-secondary)] mb-1.5">표시 이름</label>
            <input
              type="text"
              value={displayName}
              onChange={(e) => setDisplayName(e.target.value)}
              className="w-full px-3.5 py-2.5 text-sm bg-[var(--color-bg)] input-ring"
              placeholder="팀원에게 보여질 이름"
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-[var(--color-text-secondary)] mb-1.5">새 비밀번호</label>
            <input
              type="password"
              value={newPassword}
              onChange={(e) => setNewPassword(e.target.value)}
              className="w-full px-3.5 py-2.5 text-sm bg-[var(--color-bg)] input-ring"
              placeholder="변경하지 않으려면 비워두세요"
            />
          </div>
          {newPassword && (
            <div>
              <label className="block text-sm font-medium text-[var(--color-text-secondary)] mb-1.5">비밀번호 확인</label>
              <input
                type="password"
                value={confirmPassword}
                onChange={(e) => setConfirmPassword(e.target.value)}
                className="w-full px-3.5 py-2.5 text-sm bg-[var(--color-bg)] input-ring"
                placeholder="비밀번호를 다시 입력하세요"
              />
            </div>
          )}

          {profileError && (
            <div className="text-sm text-[var(--color-error)] bg-red-50 px-3 py-2 rounded-lg">
              {profileError}
            </div>
          )}

          <div className="pt-2">
            <button
              type="submit"
              disabled={savingProfile}
              className="py-2.5 px-6 text-sm btn-primary cursor-pointer"
            >
              {profileSaved ? '저장됨' : savingProfile ? '저장 중...' : '저장'}
            </button>
          </div>
        </form>
      </section>
    </div>
  );
}
```

```tsx
import { Routes, Route, Navigate } from 'react-router-dom';
import { useAuthStore } from './stores/authStore';
import Layout from './components/Layout';
import LoginPage from './pages/LoginPage';
import RegisterPage from './pages/RegisterPage';
import DashboardPage from './pages/DashboardPage';
import FundDetailPage from './pages/FundDetailPage';
import LpMasterPage from './pages/LpMasterPage';
import SettingsPage from './pages/SettingsPage';

function ProtectedRoute({ children }: { children: React.ReactNode }) {
  const token = useAuthStore((s) => s.token);
  if (!token) return <Navigate to="/login" replace />;
  return <>{children}</>;
}

export default function App() {
  return (
    <Routes>
      {/* Public routes */}
      <Route path="/login" element={<LoginPage />} />
      <Route path="/register" element={<RegisterPage />} />

      {/* Protected routes */}
      <Route
        path="/"
        element={
          <ProtectedRoute>
            <Layout />
          </ProtectedRoute>
        }
      >
        <Route index element={<DashboardPage />} />
        <Route path="fund/:id" element={<FundDetailPage />} />
        <Route path="lps" element={<LpMasterPage />} />
        <Route path="settings" element={<SettingsPage />} />
      </Route>

      {/* Fallback */}
      <Route path="*" element={<Navigate to="/" replace />} />
    </Routes>
  );
}
```
